lambda_gc <- function(chisq) {
  median(as.numeric(chisq), na.rm = TRUE) / qchisq(0.5, 1)
}

qq_df <- function(data, panel) {
  out <- lapply(names(p_map), function(method) {
    p <- sort(as.numeric(data[[p_map[[method]]]]))
    p <- p[is.finite(p) & p > 0 & p <= 1]
    n <- length(p)
    data.frame(
      panel = panel,
      method = method,
      expected = -log10((seq_len(n) - 0.5) / n),
      observed = -log10(p),
      stringsAsFactors = FALSE
    )
  })
  d <- do.call(rbind, out)
  d$method <- factor(d$method, levels = names(method_cols))
  d
}

The manuscript simulation used 1,000 replicates of 50,000 individuals, PC1 through PC5, and a common pool of 2,500 cases. Ancestry shifted both allele frequency and disease risk. Matched P values were score tests with the genotype coefficient set to zero.
